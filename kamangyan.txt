    <!-- 2. MULTI-PAGE CONTROLLER LOGIC FOR ASSIGNMENT REQUIREMENT -->
    <script>
        function switchAssignmentPage(targetViewId) {
            // Itago ang kasalukuyang active na view
            document.querySelectorAll('.assignment-page-view').forEach(view => {
                view.classList.remove('active-view');
            });

            // Ipakita ang bagong target na view
            const activeTargetView = document.getElementById(targetViewId);
            if(activeTargetView) {
                activeTargetView.classList.add('active-view');
                
                // Awtomatikong i-scroll sa pinaka-itaas para sa bagong page layout feel
                window.scrollTo({ top: 0, behavior: 'instant' });
                
                // Refresh ang ScrollTrigger calculation dahil nagbago ang taas ng page element contents
                ScrollTrigger.refresh();
                
                // Patakbuhin muli ang custom transition load ng GSAP para sa dynamic view switching appearance
                gsap.fromTo(activeTargetView.children, 
                    { opacity: 0, y: 20 },
                    { opacity: 1, y: 0, duration: 0.6, ease: "power2.out", stagger: 0.1 }
                );
            }

            // Baguhin ang status indicators sa sitemap tracking menu
            document.querySelectorAll('.nav-link-btn').forEach(btn => {
                btn.classList.remove('active-nav');
            });
            
            const activeNavBtn = document.getElementById('nav-' + targetViewId);
            if(activeNavBtn) {
                activeNavBtn.classList.add('active-nav');
            }
        }
    </script>

    <!-- COUNT-UP TIMER LOGIC FOR WEDDING DATE -->
    <script>
        const weddingDate = new Date("April 23, 2026 15:00:00").getTime();

        function updateCountdown() {
            const now = new Date().getTime();
            const distance = now - weddingDate;

            const days = Math.floor(distance / (1000 * 60 * 60 * 24));
            const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
            const seconds = Math.floor((distance % (1000 * 60)) / 1000);

            document.getElementById("days").innerHTML = days < 10 ? "0" + days : days;
            document.getElementById("hours").innerHTML = hours < 10 ? "0" + hours : hours;
            document.getElementById("minutes").innerHTML = minutes < 10 ? "0" + minutes : minutes;
            document.getElementById("seconds").innerHTML = seconds < 10 ? "0" + seconds : seconds;
        }
